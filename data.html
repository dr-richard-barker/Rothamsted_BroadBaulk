Programmatic Access (Python) 

import pandas as pd

# URL direct to the CSV endpoint from DOI resolver
url = 'https://doi.org/10.23637/rbk1-yld6822-01'

# Load dataset directly into a DataFrame
df = pd.read_csv(url)

# View first records
print(df.head())

Programmatic Access (R) 

library(readr)

# Read directly from DOI
df <- read_csv('https://doi.org/10.23637/rbk1-yld6822-01')

# View structure
str(df)
